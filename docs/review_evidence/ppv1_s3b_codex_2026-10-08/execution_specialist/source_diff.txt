+def parse_model_proposals(
+    response_text: str,
+    *,
+    allowed_hypothesis_ids: list[str] | set[str] | frozenset[str],
+) -> dict:
+    """解析模型响应的 proposals[](每项一条独立反应假设)。
+
+    - response_text 非 JSON ⇒ structure_unparseable(**结构失败**,
+      不进格式反馈 —— 不能以格式反馈改分子再试到通过);
+    - 顶层必须是 dict 且含 "proposals" list(缺字段/坏类型 ⇒ 格式
+      错误,可进有界反馈);
+    - 每项 proposal 必须是 dict:
+      * 引用冻结池假设:`hypothesis_id` ∈ allowed(备选反应假设引
+        稳定 ID;模型 rank 不把选择升成 recorded —— 解析结果只是
+        提案引用);
+      * novel 池外提案:必须携带 `proposal_context`(dict,含
+        `reference_hypothesis_ids`(可空,池内参考)与 `output_origin`
+        (明示来源,如 "novel_outside_pool"),不伪造来源);
+      * 共同产物与备选不混:proposal 内的 `coproducts`(list)是
+        **该假设内部**的共同产物声明,不作为独立提案出现(发现
+        顶层的 coproduct 形状混淆 ⇒ syntax 契约错误);
+    - `rank_hypotheses`(可选,list)引用的 id 必须在提案集合内
+      (引稳定 ID;排序是建议次序,不是已核实状态);
+    - 空列表/弃权合法:proposals=[] ⇒ abstained=True;
+    - 返回 {proposals, rank_hypotheses, abstained, protocol_version}。
+    """
+    if not isinstance(response_text, str):
+        _format_error(
+            FORMAT_ERROR_JSON_TYPE,
+            f"响应必须是 JSON 文本,得到 {type(response_text).__name__}")
+    try:
+        payload = json.loads(response_text)
+    except (ValueError, TypeError) as exc:
+        _format_error(
+            FORMAT_ERROR_STRUCTURE_UNPARSEABLE,
+            f"响应不可解析为 JSON(结构失败,不进格式反馈;不能以反馈"
+            f"改分子再试到通过):{str(exc)[:160]}")
+    if not isinstance(payload, dict):
+        _format_error(
+            FORMAT_ERROR_JSON_TYPE,
+            f"响应顶层必须是 dict,得到 {type(payload).__name__}")
+    if "proposals" not in payload:
+        _format_error(
+            FORMAT_ERROR_JSON_FIELD,
+            "响应缺 proposals 字段(每项一条独立反应假设;空列表/"
+            "弃权合法)")
+    proposals_raw = payload["proposals"]
+    if not isinstance(proposals_raw, list):
+        _format_error(
+            FORMAT_ERROR_JSON_TYPE,
+            f"proposals 必须是 list,得到 {type(proposals_raw).__name__}")
+    allowed = set(allowed_hypothesis_ids or [])
+    proposals: list[dict] = []
+    seen_ids: set[str] = set()
+    for i, item in enumerate(proposals_raw):
+        if not isinstance(item, dict):
+            _format_error(
+                FORMAT_ERROR_JSON_TYPE,
+                f"proposals[{i}] 必须是 dict,得到 {type(item).__name__}",
+                {"index": i})
+        if "coproducts" in item and "hypothesis_id" not in item:
+            _format_error(
+                FORMAT_ERROR_SYNTAX,
+                f"proposals[{i}] 携带 coproducts 但不是假设提案(共同"
+                "产物是假设内部声明,不作为独立提案;备选与共同产物"
+                "不混队)", {"index": i})
+        hypothesis_id = str(item.get("hypothesis_id") or "")
+        if not hypothesis_id:
+            _format_error(
+                FORMAT_ERROR_JSON_FIELD,
+                f"proposals[{i}] 缺 hypothesis_id(引用冻结池稳定 ID,"
+                "或 novel 时经 proposal_context 声明)", {"index": i})
+        if hypothesis_id in seen_ids:
+            _format_error(
+                FORMAT_ERROR_SYNTAX,
+                f"proposals[{i}] 重复引用 {hypothesis_id}(幂等引用,"
+                "不重复提案)", {"index": i, "hypothesis_id": hypothesis_id})
+        seen_ids.add(hypothesis_id)
+        entry: dict = {"hypothesis_id": hypothesis_id}
+        if hypothesis_id in allowed:
+            entry["origin"] = "frozen_pool"
+        else:
+            context = item.get("proposal_context")
+            if not isinstance(context, dict):
+                _format_error(
+                    FORMAT_ERROR_JSON_FIELD,
+                    f"proposals[{i}] 引用池外假设 {hypothesis_id} 但缺"
+                    " proposal_context(novel 有界:reference_hypothesis_"
+                    "ids/output_origin 明示,不伪造来源)",
+                    {"index": i, "hypothesis_id": hypothesis_id})
+            refs = context.get("reference_hypothesis_ids")
+            if not isinstance(refs, list) or \
+                    any(str(r) not in allowed for r in refs):
+                _format_error(
+                    FORMAT_ERROR_ALLOWED_ID,
+                    f"proposals[{i}] 的 proposal_context.reference_"
+                    "hypothesis_ids 必须是冻结池 ID 子集",
+                    {"index": i, "hypothesis_id": hypothesis_id,
+                     "reference_hypothesis_ids": refs})
+            if not str(context.get("output_origin") or ""):
+                _format_error(
+                    FORMAT_ERROR_JSON_FIELD,
+                    f"proposals[{i}] 的 proposal_context 缺 output_origin"
+                    "(明示来源)",
+                    {"index": i, "hypothesis_id": hypothesis_id})
+            entry["origin"] = "novel"
+            entry["proposal_context"] = dict(context)
+        coproducts = item.get("coproducts")
+        if coproducts is not None and not isinstance(coproducts, list):
+            _format_error(
+                FORMAT_ERROR_JSON_TYPE,
+                f"proposals[{i}] 的 coproducts 必须是 list(假设内部共同"
+                "产物声明)", {"index": i})
+        if coproducts is not None:
+            entry["coproducts"] = [str(c) for c in coproducts]
+        proposals.append(entry)
+    rank_raw = payload.get("rank_hypotheses")
+    if rank_raw is None:
+        rank_raw = [p["hypothesis_id"] for p in proposals]
+    if not isinstance(rank_raw, list):
+        _format_error(
+            FORMAT_ERROR_JSON_TYPE,
+            f"rank_hypotheses 必须是 list,得到 {type(rank_raw).__name__}")
+    rank: list[str] = []
+    for i, rid in enumerate(rank_raw):
+        rid = str(rid)
+        if rid not in seen_ids:
+            _format_error(
+                FORMAT_ERROR_ALLOWED_ID,
+                f"rank_hypotheses[{i}] 引用未在提案集合内的 {rid}"
+                "(rank 引稳定 ID)",
+                {"index": i, "hypothesis_id": rid})
+        rank.append(rid)
+    return {"proposals": proposals, "rank_hypotheses": rank,
+            "abstained": not proposals,
+            "protocol_version": MODEL_PROTOCOL_VERSION}
+

+def feedback_policy_from_ledger(ledger: SearchLedger,
+                                run_id: str) -> dict:
+    """从 run 的冻结策略块读取格式反馈策略(严格消费;块缺/坏 ⇒
+    SearchRecoveryError 由 _limits 同源包装,不补默认)。"""
+    block = ledger.search_policy_block(run_id)
+    policy = (block or {}).get("format_feedback_policy") \
+        if isinstance(block, dict) else None
+    try:
+        return validate_frozen_format_feedback_policy(policy)
+    except ValueError as exc:
+        raise ModelProtocolError(
+            f"run {run_id} 的冻结格式反馈策略块非法(严格核验,不补"
+            f"默认、不猜迁移):{exc}",
+            {"run_id": run_id, "kind": "format_feedback_policy_corrupt",
+             "reason": str(exc)[:300]}) from exc
+
+
+def format_feedback_rounds_used(ledger: SearchLedger,
+                                action_id: str) -> int:
+    """同 action 已发生的格式反馈轮数(持久事实源 = model_requests
+    行;不是内存计数,内存重置仍从记录复算)。"""
+    row = ledger.conn.execute(
+        "SELECT COUNT(*) FROM model_requests WHERE action_id = ? AND"
+        " purpose = ?", (action_id, PURPOSE_FORMAT_FEEDBACK)).fetchone()
+    return int(row[0])
+
+
+def register_format_feedback_round(
+    ledger: SearchLedger,
+    *,
+    run_id: str,
+    action_id: str,
+    attempt_id: str,
+    error: ModelProtocolError,
+    response_excerpt: str,
+    provider_kind: str = PROVIDER_KIND_SCRIPTED,
+) -> dict:
+    """登记一轮有界格式反馈(每轮持久记录输入/错误类别/反馈版本/
+    次数;受独立有限策略约束;超限 ⇒ 结构化拒,结构失败保持)。
+
+    - 只有 FEEDBACK_ELIGIBLE 类错误可登记(not_json 的
+      structure_unparseable 在调用侧直接记结构失败,不经本入口);
+    - 轮数 = 同 action 既有 format_feedback 请求数;≥ 策略上限 ⇒
+      SearchBudgetExhausted(reason_category=budget_limit)—— 反馈停,
+      上一次解析失败作为结构失败保持;
+    - scripted/fake 恒 network 0;real 走 register_model_request 的
+      即时占额(真实请求额度 0 ⇒ 登记被拒,网络发送被阻止;传入
+      模型对象不自动补预算);
+    - request_key 派生自 (action, 轮次) —— 幂等;payload 持久记录
+      错误类别/版本/次数/响应摘录。
+    """
+    if provider_kind not in PROVIDER_KINDS:
+        raise ModelProtocolError(
+            f"provider_kind 非法:{provider_kind!r}",
+            {"kind": "invalid_provider_kind",
+             "provider_kind": provider_kind})
+    kind = str(error.detail.get("kind") or "")
+    if kind not in FEEDBACK_ELIGIBLE:
+        eligible = sorted(FEEDBACK_ELIGIBLE)
+        raise ModelProtocolError(
+            f"错误类别 {kind!r} 不可进格式反馈(仅 {eligible};"
+            "structure_unparseable 记结构失败)",
+            {"kind": "feedback_ineligible_error", "error_kind": kind})
+    policy = feedback_policy_from_ledger(ledger, run_id)
+    rounds_used = format_feedback_rounds_used(ledger, action_id)
+    max_rounds = int(policy["max_format_feedback_rounds"])
+    if rounds_used >= max_rounds:
+        raise SearchBudgetExhausted(
+            f"格式反馈轮数超上限(同任务 JSON 修复轮数 {rounds_used} ≥ "
+            f"max_format_feedback_rounds={max_rounds};结构失败保持,"
+            "不以反馈重试到通过;该轮数不是 hypothesis 语义 attempt)",
+            {"kind": "format_feedback_budget_exhausted",
+             "limit_key": "max_format_feedback_rounds",
+             "max": max_rounds, "used": rounds_used,
+             "policy_version": FORMAT_FEEDBACK_POLICY_VERSION})
+    round_index = rounds_used + 1
+    payload = {
+        "feedback_version": FORMAT_FEEDBACK_POLICY_VERSION,
+        "model_protocol_version": MODEL_PROTOCOL_VERSION,
+        "round": round_index,
+        "error_kind": kind,
+        "error_detail": dict(error.detail),
+        "response_excerpt": str(response_excerpt)[:2000],
+        "recorded_at": now_iso(),
+    }
+    return ledger.register_model_request(
+        run_id=run_id, action_id=action_id, attempt_id=attempt_id,
+        provider_kind=provider_kind,
+        request_key=f"format-feedback:{action_id}:{round_index}",
+        purpose=PURPOSE_FORMAT_FEEDBACK, payload=payload)

+def build_cyclic_dependency(spec: dict) -> Dependency:
+    """H04 共底物循环依赖构建(§5.2:有启动供应/再生支持不能自动阻断;
+    缺支持标 unresolved_initial_supply,按显式、版本化策略处理,不称
+    化学不可能)。
+
+    spec: {"dependency_id", "kind"=cosubstrate_supply, "statement",
+    "cyclic": bool, "initial_supply": "supported"|"missing",
+    "source_refs": [...]}
+    - initial_supply=missing ⇒ status=unresolved,statement 带
+      unresolved_initial_supply 标记(audit 依赖轴 unknown ⇒ R5
+      retain_conditional,不 reject;不能称化学不可能);
+    - supported ⇒ resolution_log 记 initial_supply 轴 supported
+      (Dependency 从轨迹同源重推导 status;不自动阻断)。
+    """
+    if not isinstance(spec, dict) or not spec.get("dependency_id"):
+        raise ProductionContractError(
+            "依赖 spec 缺 dependency_id", {"kind": "dependency_spec_invalid"})
+    dep_id = str(spec["dependency_id"])
+    kind = str(spec.get("kind") or "cosubstrate_supply")
+    statement = str(spec.get("statement") or "")
+    initial_supply = str(spec.get("initial_supply") or "")
+    if initial_supply not in ("supported", "missing"):
+        raise ProductionContractError(
+            f"依赖 {dep_id} 的 initial_supply 非法:{initial_supply!r}"
+            "(supported|missing;显式声明,不猜)",
+            {"kind": "dependency_initial_supply_invalid",
+             "dependency_id": dep_id})
+    resolution_log: list[dict] = []
+    status = "unresolved"
+    if initial_supply == "supported":
+        resolution_log.append({
+            "support_axis": SUPPORT_AXIS_INITIAL_SUPPLY,
+            "resolution": "supported",
+            "source_ref": str((spec.get("source_refs") or [""])[0]),
+        })
+        status = "supported"
+    else:
+        statement = (statement + ";" if statement else "") + (
+            "unresolved_initial_supply(共底物循环缺启动/再生供给支持;"
+            "按显式、版本化策略处理,不称化学不可能)")
+    return Dependency(
+        dependency_id=dep_id, kind=kind, statement=statement,
+        status=status,
+        source_refs=tuple(str(r) for r in (spec.get("source_refs") or [])),
+        resolution_log=tuple(resolution_log))

+def prepare_hypothesis(
+    spec: dict,
+    *,
+    parent_path_id: str,
+    substrate_state,
+    source_refs: list[dict] | None = None,
+    transformation_ref: str = "",
+) -> dict:
+    """池条目 spec → 冻结假设 + 状态集 + 工具输入(单一构建源:调度
+    (取 hypothesis_id 进 action 身份)与执行(重建同一对象)共用,
+    同输入同 ID,不各写第二套派生)。
+
+    spec 契约(缺/坏 ⇒ ProductionContractError,零写入):
+    - product_smiles:主产物(工具已确定,引用不改);
+    - coproducts:共同产物 SMILES 列表(假设内部共同产物,不与备选
+      混队);
+    - edits/donor/alignment/expectation:AtomTool 编辑协议输入
+      (check_edit_against_donor 的契约形状);
+    - requirements:必要要求 dict 列表(build_hypothesis 契约);
+    - dependencies:依赖 spec 列表(H04,经 build_cyclic_dependency);
+    - participants:可选显式参与物覆盖(系数/角色;默认底物 1 + 主
+      产物 1 + 共同产物逐条,系数按声明或 unknown)。
+    返回 {hypothesis, states, product_states, tool_inputs, spec_digest}。
+    """
+    if not isinstance(spec, dict):
+        raise ProductionContractError(
+            f"池条目 spec 必须是 dict,得到 {type(spec).__name__}",
+            {"kind": "spec_invalid_type"})
+    product_smiles = str(spec.get("product_smiles") or "")
+    if not product_smiles:
+        raise ProductionContractError(
+            "池条目缺 product_smiles(工具已确定的主产物,只能引用)",
+            {"kind": "spec_missing_product"})
+    coproducts = spec.get("coproducts") or []
+    if not isinstance(coproducts, list):
+        raise ProductionContractError(
+            "coproducts 必须是 list(假设内部共同产物)",
+            {"kind": "spec_coproducts_invalid"})
+    for field in ("edits", "donor", "alignment", "expectation"):
+        value = spec.get(field)
+        if field == "edits" and not isinstance(value, list):
+            raise ProductionContractError(
+                f"spec.edits 必须是 list(冻结 AtomTool 编辑协议),得到 "
+                f"{type(value).__name__}",
+                {"kind": "spec_edits_invalid"})
+        if field != "edits" and not isinstance(value, dict):
+            raise ProductionContractError(
+                f"spec.{field} 必须是 dict(供体诊断输入契约),得到 "
+                f"{type(value).__name__}",
+                {"kind": f"spec_{field}_invalid"})
+    refs = source_refs or [{"record": "synthetic-pool-spec"}]
+    product_state = build_product_state(product_smiles, source_refs=refs)
+    coproduct_states = [
+        build_product_state(str(c), source_refs=refs) for c in coproducts]
+    dependencies = [
+        build_cyclic_dependency(d)
+        for d in (spec.get("dependencies") or [])]
+    participants_spec = spec.get("participants")
+    if participants_spec is not None:
+        participants = [
+            Participant(**dict(p)) for p in participants_spec]
+    else:
+        coefficient = spec.get("coefficient")
+        coefficient_status = ("known" if coefficient is not None
+                              else "unknown")
+        participants = [
+            Participant(
+                state_ref=substrate_state.state_id, side="reactant",
+                role="primary_substrate",
+                coefficient=(coefficient if coefficient is not None else None),
+                coefficient_status=coefficient_status),
+            Participant(
+                state_ref=product_state.state_id, side="product",
+                role="principal_product",
+                coefficient=(coefficient if coefficient is not None else None),
+                coefficient_status=coefficient_status),
+        ]
+        for state in coproduct_states:
+            participants.append(Participant(
+                state_ref=state.state_id, side="product", role="coproduct",
+                coefficient=None, coefficient_status="unknown"))
+    requirements = [
+        dict(r) for r in (spec.get("requirements") or [])]
+    hypothesis = build_hypothesis(
+        parent_path_id=parent_path_id,
+        substrate_state_ref=substrate_state.state_id,
+        participants=participants,
+        participant_list_status=str(spec.get("participant_list_status")
+                                    or "partial"),
+        step_scope=str(spec.get("step_scope") or "net_reaction"),
+        transformation_ref=transformation_ref,
+        requirements=requirements,
+        dependency_refs=[d.dependency_id for d in dependencies],
+        origin_group_key=str(spec.get("origin_group_key") or ""))
+    states = {substrate_state.state_id: substrate_state,
+              product_state.state_id: product_state}
+    for state in coproduct_states:
+        states[state.state_id] = state
+    tool_inputs = {
+        "edits": list(spec["edits"]),
+        "donor": dict(spec["donor"]),
+        "alignment": dict(spec["alignment"]),
+        "expectation": dict(spec["expectation"]),
+    }
+    return {
+        "hypothesis": hypothesis,
+        "states": states,
+        "product_states": [product_state] + coproduct_states,
+        "dependencies": dependencies,
+        "tool_inputs": tool_inputs,
+        "spec_digest": content_hash(spec),
+        "hypothesis_dict": hypothesis.to_dict(),
+    }

+            (run_id, "max_alternative_hypotheses_per_parent",

+    max_select = None
+    if budget is not None:

+                "budget 视图必须是 dict", {"kind": "budget_view_invalid"})

+                    {"kind": "budget_max_select_invalid"})

+    if max_select is not None:
+        selectable = min(selectable, max_select)

+        enumerate(pool or []),
+        key=lambda pair: (
+            (pair[1].get("rank") is None, pair[1].get("rank") or 0,
+             str(pair[1].get("hypothesis_id", "")), pair[0])))
+    digest = frozen_pool_digest(pool or [])
+    selected: list[dict] = []
+    deferred: list[dict] = []
+    blocked: list[dict] = []

+                "reason": "depth_limit",
+                "detail": {"depth": depth,
+                           "max_search_depth":
+                               int(limits["max_search_depth"])}})

+            "context_ref": context_ref,
+            "parent_search_revision": int(parent_search_revision),
+            "pool_digest": digest, "pool_size": len(pool or []),
+            "selected": selected, "deferred": deferred, "blocked": blocked,
+            "remaining_alternatives": remaining_alternatives,
+            "alternatives_max": alternatives_max,
+        }
+    for _, entry in ordered:
+        hypothesis_id = str(entry["hypothesis_id"])
+        # 每条目的 context_ref(分支隔离:带自身要求/依赖的假设经
+        # derive_child_context 派生专属上下文;缺省用调用的父上下文)
+        entry_context = str(entry.get("payload", {}).get("context_ref")
+                            or context_ref)

+        task_payload["frozen_pool_digest"] = digest
+        task_payload.setdefault("source_event_id", source_event_id)

+        if will_select:
+            enqueue = ledger.enqueue_action(
+                run_id=run_id, parent_path_id=parent_path_id,
+                context_ref=entry_context, subject_ref=hypothesis_id,
+                action_kind=ACTION_KIND_EXPAND_HYPOTHESIS,
+                parent_search_revision=parent_search_revision,
+                policy_version=policy_version, payload=task_payload,
+                depth=depth, source_event_id=source_event_id)
+        else:
+            reason = ("selection_cap" if max_select is not None
+                      and remaining_alternatives > taken
+                      else "alternatives_budget")
+            task_payload["deferred_type"] = DEFERRED_TYPE_ALTERNATIVE
+            task_payload["defer_reason"] = reason
+            enqueue = ledger.enqueue_action(
+                run_id=run_id, parent_path_id=parent_path_id,
+                context_ref=entry_context, subject_ref=hypothesis_id,
+                action_kind=ACTION_KIND_EXPAND_HYPOTHESIS,
+                parent_search_revision=parent_search_revision,
+                policy_version=policy_version, payload=task_payload,
+                depth=depth, source_event_id=source_event_id,
+                initial_status=ACTION_STATUS_DEFERRED,
+                defer_reason=reason, defer_trigger=defer_trigger)
+        if not enqueue.get("created"):
+            status = enqueue.get("status")
+            if enqueue.get("reason") == "depth_limit":
+                blocked.append({"hypothesis_id": hypothesis_id,
+                                "reason": "depth_limit"})
+                continue
+            if status == ACTION_STATUS_DONE:
+                blocked.append({"hypothesis_id": hypothesis_id,
+                                "reason": "already_expanded",
+                                "action_id": enqueue.get("action_id")})
+                continue

+                        "action_id": enqueue.get("action_id"),
+                        "status": status,
+                        "reason": "already_enqueued"}
+            if status == ACTION_STATUS_DEFERRED:
+                deferred.append(existing)
+            else:

+            selected.append({"hypothesis_id": hypothesis_id,
+                             "action_id": enqueue["action_id"],
+                             "status": ACTION_STATUS_PENDING})
+            taken += 1
+        else:
+            deferred.append({
+                "hypothesis_id": hypothesis_id,
+                "action_id": enqueue["action_id"],
+                "status": ACTION_STATUS_DEFERRED,
+                "deferred_type": DEFERRED_TYPE_ALTERNATIVE,
+                "reason": str(task_payload.get("defer_reason")),
+            })
+    return {
+        "run_id": run_id, "parent_path_id": parent_path_id,
+        "context_ref": context_ref,
+        "parent_search_revision": int(parent_search_revision),
+        "pool_digest": digest, "pool_size": len(pool or []),
+        "selected": selected, "deferred": deferred, "blocked": blocked,
+        "remaining_alternatives": remaining_alternatives,
+        "alternatives_max": alternatives_max,
+    }

+def record_expansion_outcome(
+    ledger: SearchLedger, attempt_id: str, *,
+    outcome: dict, evidence_refs: list[str] | None = None,
+    worker_id: str,
+) -> dict:
+    """持久化一次展开尝试的**非事件结论**(E01 的 unresolved_
+    continuation、工具/契约方法失败等;事件类结算走
+    submit_action_event + complete_action,不经本入口)。
+
+    - attempt 须存在、归属本 run 且 open,且其 action 处于 claimed、
+      认领者为 worker(执行权闭合;陈旧 worker 不可落账);
+    - outcome.kind 枚举(未知 ⇒ 契约拒);method 类失败
+      (method_failed/contract_aborted)⇒ attempt failed(方法失败不是
+      化学判定);其余 ⇒ recorded;
+    - 结论 artifact 先写(task_key 绑定 action 冻结任务键;不可覆盖),
+      结算经 complete_action(outcome 类;S3A-02 完整权威核验);
+    - outcome_ref 返回给调用方(事件/账本关联的持久引用)。
+    """
+    if not isinstance(outcome, dict) or \
+            outcome.get("kind") not in OUTCOME_KINDS:
+        raise ScheduleContractError(
+            f"outcome.kind 非法:{(outcome or {}).get('kind')!r}"
+            f"(允许 {sorted(OUTCOME_KINDS)})",
+            {"kind": "invalid_outcome_kind"})
+    row = ledger.conn.execute(
+        "SELECT * FROM search_attempts WHERE expansion_attempt_id = ?",
+        (attempt_id,)).fetchone()
+    if row is None:
+        raise ScheduleContractError(
+            f"attempt 不存在:{attempt_id}",
+            {"attempt_id": attempt_id, "kind": "attempt_missing"})
+    action_id = row["action_id"]
+    action = ledger.conn.execute(
+        "SELECT * FROM search_actions WHERE action_id = ?",
+        (action_id,)).fetchone()
+    if action is None or action["run_id"] != row["run_id"]:
+        raise ScheduleContractError(
+            f"attempt {attempt_id} 引用的 action 缺失或异属(执行引用"
+            "闭合失败)",
+            {"attempt_id": attempt_id, "action_id": action_id,
+             "kind": "attempt_action_binding_mismatch"})
+    if action["status"] != "claimed":
+        raise ScheduleContractError(
+            f"action {action_id} 状态为 {action['status']}(非事件结论"
+            "只在认领执行权下落账)",
+            {"attempt_id": attempt_id, "action_id": action_id,
+             "kind": "not_claimed", "status": action["status"]})
+    if action["claimed_by"] != worker_id:
+        raise ScheduleContractError(
+            f"action {action_id} 由 {action['claimed_by']!r} 认领,"
+            f"{worker_id!r} 不可越权落账尝试结论",
+            {"attempt_id": attempt_id, "action_id": action_id,
+             "kind": "owner_mismatch"})
+    if row["status"] != ATTEMPT_STATUS_OPEN:
+        raise ScheduleContractError(
+            f"attempt {attempt_id} 状态为 {row['status']}(已落账的尝试"
+            "不可再记非事件结论)",
+            {"attempt_id": attempt_id, "kind": "attempt_not_open",
+             "status": row["status"]})
+    payload = {
+        "outcome_version": EXPANSION_OUTCOME_VERSION,
+        "attempt_id": attempt_id, "action_id": action_id,
+        "run_id": row["run_id"], "hypothesis_id": row["hypothesis_id"],
+        "outcome": dict(outcome),
+        "evidence_refs": [str(r) for r in (evidence_refs or [])],
+        "recorded_at": now_iso(),
+    }
+    task_key = action["task_key"] or stable_id("soutc", action_id)
+    if ledger.artifact_store is not None:
+        ledger.artifact_store.write(task_key, "expansion_outcome", payload)
+    outcome_ref = content_hash(payload)
+    attempt_status = (ATTEMPT_STATUS_FAILED
+                      if outcome["kind"] in _OUTCOME_FAILED_KINDS
+                      else ATTEMPT_STATUS_RECORDED)
+    return ledger.complete_action(
+        action_id, worker_id=worker_id,
+        outcome_kind=COMMIT_KIND_OUTCOME, outcome_ref=outcome_ref,
+        attempt_status=attempt_status)

+def activate_deferred(
+    ledger: SearchLedger,
+    *,
+    run_id: str,
+    parent_path_id: str,
+    trigger: str,
+    expected_parent_revision: int,
+    deferred_type: str | None = None,
+) -> dict:
+    """激活 deferred(两类分别;幂等;trigger/expected_parent_revision
+    校验;预算不足留独立未展开原因;不重复安排同 action)。
+
+    - deferred_type=None ⇒ 父路径全部 deferred;指定
+      coproduct_deferred / alternative_reaction_deferred ⇒ 只激活该类
+      (两类不混队);
+    - 每项经 ledger.reactivate_deferred_action 同一判据:p ok/
+      claimed/done ⇒ unchanged already_active(重复触发幂等);旧
+      revision ⇒ unchanged stale_parent_revision;预算不足 ⇒
+      unchanged budget_exhausted(独立未展开原因,不改状态);trigger
+      与冻结激活条件不一致 ⇒ 逐项结构化冲突(冲突不是成功激活);
+    - 激活只把 deferred → pending(执行仍走 claim 的权威预算事务);
+     不重建任务输入、不覆盖冻结身份。
+    """
+    if deferred_type is not None and deferred_type not in DEFERRED_TYPES:
+        raise ScheduleContractError(
+            f"deferred_type 非法:{deferred_type!r}(允许 "
+            f"{sorted(DEFERRED_TYPES)};None = 两类全部)",
+            {"kind": "invalid_deferred_type", "deferred_type": deferred_type})
+    rows = ledger.conn.execute(
+        "SELECT * FROM search_actions WHERE run_id = ? AND"
+        " parent_path_id = ? ORDER BY created_at, action_id",
+        (run_id, parent_path_id)).fetchall()
+    items = []
+    for row in rows:
+        payload = json.loads(row["payload"])
+        item = dict(row)
+        item["payload"] = payload
+        if deferred_type is None:
+            if not str(payload.get("deferred_type") or ""):
+                continue  # 非 deferred 载荷的 action 不参与激活
+        elif str(payload.get("deferred_type") or "") != deferred_type:
+            continue
+        items.append(item)
+    activated_ids: list[str] = []
+    unchanged: list[dict] = []
+    conflicts: list[dict] = []
+    for item in items:
+        try:
+            out = ledger.reactivate_deferred_action(
+                item["action_id"], trigger=trigger,
+                expected_parent_revision=expected_parent_revision)
+        except SearchContractError as exc:
+            conflicts.append({"action_id": item["action_id"],
+                              "hypothesis_or_object": item["subject_ref"],
+                              "deferred_type": str(
+                                  item["payload"].get("deferred_type")
+                                  or ""),
+                              "error": type(exc).__name__,
+                              "kind": exc.detail.get("kind"),
+                              "detail": exc.detail})
+            continue
+        entry = {"action_id": item["action_id"],
+                 "hypothesis_or_object": item["subject_ref"],
+                 "deferred_type": str(
+                     item["payload"].get("deferred_type") or ""),
+                 "reason": out.get("reason")}
+        if out.get("detail"):
+            entry["detail"] = out.get("detail")
+        if out.get("activated"):
+            activated_ids.append(item["action_id"])
+        else:
+            unchanged.append(entry)
+    return {"run_id": run_id, "parent_path_id": parent_path_id,
+            "deferred_type": deferred_type, "trigger": str(trigger),
+            "expected_parent_revision": int(expected_parent_revision),
+            "activated_ids": activated_ids, "unchanged": unchanged,
+            "conflicts": conflicts, "matched": len(items)}
+

+def enqueue_scenario_transition(
+    ledger: SearchLedger,
+    *,
+    run_id: str,
+    parent_path_id: str,
+    context_ref: str,
+    transition: dict,
+    policy_version: str,
+    parent_search_revision: int,

+        raise ScheduleContractError(

+                {"kind": "transition_field_missing", "field": key})

+    return ledger.enqueue_action(
+        run_id=run_id, parent_path_id=parent_path_id,
+        context_ref=context_ref,
+        subject_ref=stable_id("strans", parent_path_id, context_ref,
+                               content_hash(transition)),
+        action_kind=ACTION_KIND_SCENARIO_TRANSITION,
+        parent_search_revision=parent_search_revision,
+        policy_version=policy_version, payload=payload)

+def enqueue_coproduct_followups(
+    ledger: SearchLedger,
+    *,
+    run_id: str,
+    parent_path_id: str,
+    context_ref: str,
+    event_id: str,
+    product_state_refs: list[str],
+    expansion_plan: dict,
+    policy_version: str,
+    parent_search_revision: int,
+    trigger: str = DEFERRED_TRIGGER_EXPLICIT,
+    budget_remaining_zero: bool = False,
+    substrate_smiles_by_ref: dict[str, str] | None = None,
+    depth: int = 0,
+) -> dict:
+    """事件提交后的共同产物 followup 调度(§5.1:完整事件保存全部已知
+    参与物/产物,报告 K/展开宽度或预算不足**不得删事件事实**;本入口
+    只调度 followup)。
+
+    - product_state_refs:本事件产物参与物 state_ref 全集(调用方从
+      冻结事件/提交结果取;完整事件事实不经本入口删改);
+    - 展开计划(expanded/deferred/not_tracked)引用必须是本事件产物
+      state_ref(计划引用非产物 ⇒ 契约拒,不猜映射);
+    - expanded 产物 ⇒ pending retrieve followup(子路径 = 父路径 +
+      事件 + 产物;subject = 产物 state_ref);
+    - deferred 产物(宽度/预算不足)⇒ deferred retrieve followup
+      (deferred_type=coproduct_deferred,对象 = 产物 state_ref,独立
+      未展开原因来自计划 reasons;完整事件不截断);
+    - not_tracked 产物 ⇒ 不调度(事件内事实保留);
+    - 预算不足(budget_remaining_zero)时全部产物走 deferred(
+      reason=budget_limit);
+    - retrieve followup 不预约预算(claim 才预约;检索预约按
+      max_retrievals_* 在 claim 发生)。
+    """
+    plan = expansion_plan or {}
+    known = [str(r) for r in (product_state_refs or [])]
+    known_set = set(known)
+    # 展开计划载荷形状 = ExpansionPlan.to_dict():{"entries": [
+    # {state_ref, status, reason}], ...};status ∈ expanded/deferred/
+    # not_tracked(旧形状 expanded/deferred/not_tracked 键也兼容读取)
+    entry_reasons: dict[str, str] = {}
+    expanded: list[str] = []
+    deferred_plan: list[str] = []
+    not_tracked: list[str] = []
+    for entry in (plan.get("entries") or []):
+        if not isinstance(entry, dict):
+            continue
+        ref = str(entry.get("state_ref") or "")
+        status = str(entry.get("status") or "")
+        if status == "expanded":
+            expanded.append(ref)
+        elif status == "deferred":
+            deferred_plan.append(ref)
+        elif status == "not_tracked":
+            not_tracked.append(ref)
+        if entry.get("reason"):
+            entry_reasons[ref] = str(entry["reason"])
+    expanded += [str(r) for r in (plan.get("expanded") or [])
+                 if str(r) not in expanded]
+    deferred_plan += [str(r) for r in (plan.get("deferred") or [])
+                      if str(r) not in deferred_plan
+                      and str(r) not in expanded]
+    not_tracked += [str(r) for r in (plan.get("not_tracked") or [])
+                    if str(r) not in not_tracked]
+    for label, refs in (("expanded", expanded), ("deferred", deferred_plan),
+                        ("not_tracked", not_tracked)):
+        unknown = sorted(set(refs) - known_set)
+        if unknown:
+            raise ScheduleContractError(
+                f"展开计划 {label} 引用了非本事件产物:{unknown[:5]}"
+                "(计划只调度事件产物的 followup,不猜映射)",
+                {"kind": "plan_reference_not_product",
+                 "label": label, "unknown": unknown[:10]})
+    reasons = dict(plan.get("reasons") or {})
+    scheduled: list[dict] = []
+    deferred_out: list[dict] = []
+    for ref in known:
+        if ref in not_tracked:
+            continue
+        child = child_path_id(parent_path_id, event_id, ref)
+        payload = {"task_key": "", "product_state_ref": ref,
+                   "source_event_id": event_id,
+                   "parent_path_id": parent_path_id,
+                   "context_ref": context_ref,
+                   "substrate_smiles": str(
+                       (substrate_smiles_by_ref or {}).get(ref, "")),
+                   "depth": int(depth)}
+        if ref in expanded and not budget_remaining_zero:
+            out = ledger.enqueue_action(
+                run_id=run_id, parent_path_id=child,
+                context_ref=context_ref, subject_ref=ref,
+                action_kind=ACTION_KIND_RETRIEVE,
+                parent_search_revision=parent_search_revision,
+                policy_version=policy_version, payload=payload)
+            scheduled.append({"product_state_ref": ref,
+                              "action_id": out.get("action_id"),
+                              "status": out.get("status"),
+                              "child_path_id": child})
+            continue
+        reason = (DEFER_REASON_BUDGET if budget_remaining_zero
+                  else str(entry_reasons.get(ref)
+                           or reasons.get(ref)
+                           or DEFER_REASON_PRODUCT_WIDTH))
+        payload["deferred_type"] = DEFERRED_TYPE_COPRODUCT
+        out = ledger.enqueue_action(
+            run_id=run_id, parent_path_id=child, context_ref=context_ref,
+            subject_ref=ref, action_kind=ACTION_KIND_RETRIEVE,
+            parent_search_revision=parent_search_revision,
+            policy_version=policy_version, payload=payload,
+            initial_status=ACTION_STATUS_DEFERRED, defer_reason=reason,
+            defer_trigger=trigger)
+        deferred_out.append({"product_state_ref": ref,
+                             "action_id": out.get("action_id"),
+                             "status": out.get("status"),
+                             "reason": reason,
+                             "child_path_id": child})
+    return {"run_id": run_id, "parent_path_id": parent_path_id,
+            "event_id": event_id, "scheduled": scheduled,
+            "deferred": deferred_out,
+            "not_tracked": not_tracked,
+            "note": ("完整事件事实不截断;deferred/知识缺口存在时不称"
+                     "所有路线都探索完")}

+职责:workflow.runner 的 PPV1 分流目标 —— **独立**执行/存储(不落
+legacy CN/PathwayStore/预算派生/LangGraph):
+- start:严格 root state(工厂守卫)→ 根 context(持久化 artifact,
+  消费经 from_dict 内容身份核验)→ 注册搜索 run(profile 策略块:十
+  上限 + format_feedback_policy)→ 根 retrieve 入队 → 确定性驱动循环;
+- resume:身份逐字段核对(code/资产/策略/profile/prediction)→
+  recover_run 四崩溃边界(B4 补账/B3 消费 bundle/B1 续跑/B2 披露)
+  → 继续驱动循环(同 worker 身份幂等 reclaim;不重建/不覆盖既有
+  搜索/attempt/request/提交记录);
+- 驱动循环(scripted/规则级):pending 顺序认领 → 按 action_kind 分发:
+  retrieve(池获取)/ decide(schedule_hypotheses;scripted 模型
+  proposals[] 协议 + 有界格式反馈)/ expand_hypothesis(固定生产链:
+  AtomTool→诊断→converter→audit→admission→提交;**AuditContractError
+  窄捕获记 contract_aborted 方法失败,不吞成成功**)/
+  activate_deferred(两类 deferred 激活)/ scenario_transition
+  (derive_child_context 冻结旧阶段快照,只后效,转换后 followup 用
+  新上下文);
+- 停止:队列耗尽 + stop_report(主因优先级、claimed/open/deferred
+  未探索集合);停止不是化学终点。
+
+**预算纪律**:PPV1 不从 model 对象派生任何真实请求预算(与 legacy
+的 max_llm_calls 随展开派生不同 —— 传入 model 只记录身份,额度仍是
+search_limits 的 max_real_model_requests,默认 0;真实请求登记被
+register_model_request 拒绝 ⇒ 网络发送被阻止)。scripted/fake 恒
+network_request_count=0。
+
+**计划(plan)是合成/规则级输入**(测试与验收用;不是生产数据):
+{condition_facts, scenario_ref, dependencies, pools: {substrate_smiles:
+[spec...]}, model: {provider_kind, responses: [text...]},
+scenario_transitions: [{after_hypothesis_key, transition}],
+activations: [{deferred_type, trigger, expected_parent_revision,
+parent}], schedule_budget: {max_select}}。

+def _complete_action(ledger: SearchLedger, action_id: str, *,
+                     worker_id: str, outcome: dict,
+                     evidence_refs: list[str] | None = None) -> dict:
+    """非事件完成:outcome artifact 先写 + complete_action(outcome 类;
+    S3A-02 完整权威核验)。返回 outcome_ref。"""
+    payload = {"outcome_version": "ppv1-runflow-outcome-1",
+               "action_id": action_id, "outcome": dict(outcome),
+               "evidence_refs": [str(r) for r in (evidence_refs or [])]}

+        ledger.artifact_store.write(task_key, "expansion_outcome", payload)
+    return ledger.complete_action(
+        action_id, worker_id=worker_id, outcome_kind=COMMIT_KIND_OUTCOME,
+        outcome_ref=outcome_ref)

+class _Driver:
+    """一次驱动会话(进程内;恢复后新实例从持久记录继续)。"""
+
+    def __init__(self, ledger: SearchLedger, plan: dict, run_id: str,
+                 worker_id: str, crash_after: str | None = None) -> None:
+        self.ledger = ledger
+        self.store = ledger.store
+        self.plan = plan or {}
+        self.run_id = run_id
+        self.worker_id = worker_id
+        self.crash_after = crash_after
+        self.refusals: list[dict] = []
+        self.run_error: dict | None = None
+        self.empty_pool_retrievals = 0
+        self.total_retrievals = 0
+        self.events_submitted: list[str] = []
+        self.root_path_id = stable_id("spath", run_id, "root")
+        self.model_responses = list(
+            (self.plan.get("model") or {}).get("responses") or [])
+        self.model_provider_kind = str(
+            (self.plan.get("model") or {}).get("provider_kind")
+            or PROVIDER_KIND_SCRIPTED)
+
+    # ------------------------------------------------------------ 主循环
+
+    def reconcile_followups(self) -> None:
+        """B4 补救:事件已提交、followup 调度未完成的 action(done 但
+        事件产物无任何 followup)⇒ 从权威记录(事件 + 展开计划附件)
+        幂等补调度(不重发事件、不重复扣额;queue 幂等)。转换分支:
+        存在携带本事件 pending_followup 的转换 action ⇒ 留给转换执行。
+        """
+        rows = self.ledger.conn.execute(
+            "SELECT a.action_id, a.parent_path_id, a.context_ref,"
+            " a.parent_search_revision, a.payload,"
+            " c.event_id FROM search_actions a JOIN search_commit_records"
+            " c ON c.action_id = a.action_id WHERE a.run_id = ? AND"
+            " a.action_kind = ? AND a.status = ? AND c.event_id != ''",
+            (self.run_id, ACTION_KIND_EXPAND_HYPOTHESIS,
+             "done")).fetchall()
+        for row in rows:
+            event_id = row["event_id"]
+            existing = self.ledger.conn.execute(
+                "SELECT COUNT(*) FROM search_actions WHERE run_id = ? AND"
+                " payload LIKE ?",
+                (self.run_id, f"%{event_id}%")).fetchone()[0]
+            if int(existing) > 1:  # 事件引用已出现在 followup/转换载荷
+                continue
+            plan = self.store.expansion_plan(event_id)
+            if not plan:
+                continue
+            ev = self.store.get_event(event_id)
+            if not ev:
+                continue
+            ev_payload = ev.get("payload")
+            participants = (json.loads(ev_payload).get("participants")
+                            if isinstance(ev_payload, str) and ev_payload
+                            else ev.get("participants")) or []
+            product_refs = [
+                str(p.get("state_ref")) for p in participants
+                if isinstance(p, dict) and p.get("side") == "product"]
+            smiles_by_ref = {ref: self._state_smiles(ref)
+                             for ref in product_refs}
+            enqueue_coproduct_followups(
+                self.ledger, run_id=self.run_id,
+                parent_path_id=row["parent_path_id"],
+                context_ref=row["context_ref"], event_id=event_id,
+                product_state_refs=product_refs,
+                expansion_plan=dict(plan),
+                policy_version=search_policy_version(),
+                parent_search_revision=int(
+                    row["parent_search_revision"]) + 1,
+                substrate_smiles_by_ref=smiles_by_ref,
+                depth=int((json.loads(row["payload"]) or {}).get("depth")
+                          or 1))
+
+    def _inflight_action_ids(self) -> list[str]:
+        """在途 action(pending + 本 worker 已认领未完成 —— B1 续跑:
+        同 worker 幂等 reclaim 后继续执行;其他 worker 认领的 action
+        不在驱动清单内,由其执行方续跑)。稳定次序:created_at,
+        action_id。"""
+        rows = self.ledger.conn.execute(
+            "SELECT action_id FROM search_actions WHERE run_id = ? AND"
+            " (status = ? OR (status = ? AND claimed_by = ?))"
+            " ORDER BY created_at, action_id",
+            (self.run_id, "pending", "claimed", self.worker_id)).fetchall()
+        return [str(r["action_id"]) for r in rows]
+
+    def drive(self) -> dict:
+        self.reconcile_followups()
+        activations_applied = False
+        while True:
+            pending = self._inflight_action_ids()
+            if not pending:
+                directives = self.plan.get("activations") or []
+                if not activations_applied and directives:
+                    # 显式激活指令(计划输入):入队 activate_deferred
+                    # action(幂等:done 的指令 action 复用;不重复安排)
+                    activations_applied = True
+                    for directive in directives:
+                        self._enqueue_activation(directive)
+                    continue
+                break
+            action_id = pending[0]
+            try:
+                receipt = self.ledger.claim_action(
+                    action_id, worker_id=self.worker_id)
+            except SearchBudgetExhausted as exc:
+                self.refusals.append({
+                    "reason": "budget_limit",
+                    "limit_key": exc.detail.get("limit_key"),
+                    "action_id": action_id,
+                    "kind": exc.detail.get("kind")})
+                break  # 预算拒:停止驱动(stop_report 报主因)
+            except (SearchClaimError, SearchConcurrencyError) as exc:
+                # already_claimed(并发)/锁:停止,结构化披露(SQLite
+                # busy 是可见并发结果,不当化学拒绝)
+                self.refusals.append({
+                    "reason": "claim_error", "action_id": action_id,
+                    "kind": exc.detail.get("kind")})
+                break
+            if self.crash_after == "after_claim":
+                raise SearchCrashInjection(
+                    "认领已落账、执行未开始(B1 注入点)",
+                    {"action_id": action_id})
+            if not receipt.get("claimed") and \
+                    receipt.get("status") == "done":
+                continue  # done 复用(权威核验已在 claim 内)
+            kind = receipt["action_kind"]
+            try:
+                if kind == ACTION_KIND_RETRIEVE:
+                    self._run_retrieve(receipt)
+                elif kind == ACTION_KIND_DECIDE:
+                    self._run_decide(receipt)
+                elif kind == ACTION_KIND_EXPAND_HYPOTHESIS:
+                    self._run_expand(receipt)
+                elif kind == ACTION_KIND_ACTIVATE_DEFERRED:
+                    self._run_activate(receipt)
+                elif kind == ACTION_KIND_SCENARIO_TRANSITION:
+                    self._run_transition(receipt)
+                else:
+                    raise PPV1RunError(
+                        f"未知 action_kind:{kind!r}(驱动分发缺实现)",
+                        {"kind": "unknown_action_kind",
+                         "action_kind": kind})
+            except SearchCrashInjection:
+                raise  # 崩溃注入点不是 run 错误:原样上抛(测试边界)
+            except (PPV1RunError, ModelProtocolError,
+                    SearchContractError) as exc:
+                # 模型结构失败/调度契约失败(如转换缺 source_ref)等
+                # run 级错误:如实停止(不吞、不猜;claimed action 保持
+                # 认领态计入未完成,不称 exhausted_search;错误进报告)
+                self.run_error = {
+                    "error": type(exc).__name__,
+                    "kind": exc.detail.get("kind"),
+                    "message": str(exc)[:300]}
+                break
+        return self._stop_report()
+
+    # ------------------------------------------------------------ retrieve
+
+    def _run_retrieve(self, receipt: dict) -> None:
+        payload = receipt["payload"]
+        substrate_smiles = str(payload.get("substrate_smiles") or "")
+        source_event_id = str(payload.get("source_event_id") or "")
+        substrate_depth = int(payload.get("depth") or 0)
+        specs = self._pool_for(substrate_smiles)
+        self.total_retrievals += 1
+        if not specs:
+            self.empty_pool_retrievals += 1
+            # E01:首选下游无候选 ⇒ 记 unresolved_continuation(不改
+            # 上游真假;上游事件/attempt 保留)
+            _complete_action(
+                self.ledger, receipt["action_id"], worker_id=self.worker_id,
+                outcome={"kind": OUTCOME_KIND_NO_LOCAL_CANDIDATE,
+                         "detail": {
+                             "substrate_smiles": substrate_smiles,
+                             "source_event_id": source_event_id,
+                             "unresolved_continuation": bool(source_event_id),
+                             "note": ("检索/池无候选;不反证上游;按稳定"
+                                      "次序继续可用备选")}},
+                evidence_refs=[source_event_id] if source_event_id else [])
+            return
+        digest = content_hash({"specs": specs})
+        task_key = decision_task_key(
+            run_id=self.run_id, parent_path_id=receipt["parent_path_id"],
+            context_ref=receipt["context_ref"],
+            frozen_pool_digest=digest,
+            parent_search_revision=int(
+                receipt["parent_search_revision"]),
+            policy_version=receipt["policy_version"],
+            strict_state_digest=stable_id("sfeat", substrate_smiles),
+            tool_id="atom-tool-vendored",
+            prompt_version=_plan_digest(self.plan)[:16],
+            feedback_policy_version=FORMAT_FEEDBACK_POLICY_VERSION)
+        self.ledger.enqueue_action(
+            run_id=self.run_id, parent_path_id=receipt["parent_path_id"],
+            context_ref=receipt["context_ref"],
+            subject_ref=f"pool:{digest[:16]}",
+            action_kind=ACTION_KIND_DECIDE,
+            parent_search_revision=receipt["parent_search_revision"],
+            policy_version=receipt["policy_version"],
+            payload={"task_key": task_key, "pool_digest": digest,
+                     "substrate_smiles": substrate_smiles,
+                     "context_ref": receipt["context_ref"],
+                     "parent_path_id": receipt["parent_path_id"],
+                     "depth": substrate_depth,
+                     "source_event_id": source_event_id})
+        _complete_action(
+            self.ledger, receipt["action_id"], worker_id=self.worker_id,
+            outcome={"kind": OUTCOME_KIND_POOL_DIGEST,
+                     "detail": {"substrate_smiles": substrate_smiles,
+                                "pool_digest": digest,
+                                "pool_size": len(specs)}})
+
+    def _pool_for(self, substrate_smiles: str) -> list[dict]:
+        pools = self.plan.get("pools") or {}
+        if not isinstance(pools, dict):
+            raise PPV1RunError("plan.pools 必须是 dict",
+                               {"kind": "plan_pools_invalid"})
+        specs = pools.get(substrate_smiles) or []
+        if not isinstance(specs, list):
+            raise PPV1RunError(
+                f"plan.pools[{substrate_smiles!r}] 必须是 list",
+                {"kind": "plan_pool_specs_invalid"})
+        return specs
+
+    # ------------------------------------------------------------ decide
+
+    def _run_decide(self, receipt: dict) -> None:
+        payload = receipt["payload"]
+        substrate_smiles = str(payload.get("substrate_smiles") or "")
+        substrate_depth = int(payload.get("depth") or 0)
+        specs = self._pool_for(substrate_smiles)
+        substrate = _substrate_state(substrate_smiles,
+                                     f"ppv1-substrate:{self.run_id}")
+        parent_ctx = load_context(self.ledger.artifact_store,
+                                  receipt["context_ref"])
+        prepared_entries = []
+        for index, spec in enumerate(specs):
+            key = str(spec.get("key") or f"spec-{index}")
+            prepared = prepare_hypothesis(
+                spec, parent_path_id=receipt["parent_path_id"],
+                substrate_state=substrate,
+                transformation_ref=str(spec.get("transformation_ref")
+                                        or ""))
+            prepared_entries.append((key, spec, prepared))
+        model_note: dict = {}
+        if self.plan.get("model"):
+            try:
+                model_note = self._model_proposals(receipt, prepared_entries)
+            except SearchBudgetExhausted as exc:
+                # 真实 provider 额度 0(默认):登记被拒 ⇒ 网络发送被
+                # 阻止;decide 记方法失败,搜索按预算原因停止 ——
+                # **不因传 model 对象补/派生真实请求预算**
+                self.refusals.append({
+                    "reason": "budget_limit",
+                    "limit_key": exc.detail.get("limit_key"),
+                    "action_id": receipt["action_id"],
+                    "kind": exc.detail.get("kind")})
+                _complete_action(
+                    self.ledger, receipt["action_id"],
+                    worker_id=self.worker_id,
+                    outcome={"kind": OUTCOME_KIND_METHOD_FAILED,
+                             "detail": {"error": "real_request_refused",
+                                        "note": ("真实模型请求额度不足,"
+                                                 "网络发送被阻止;传入"
+                                                 "模型对象不自动补预算")}})
+                return
+        rank_by_key = model_note.get("rank_by_key") or {}
+        entries: list[dict] = []
+        expand_depth = substrate_depth + 1
+        for index, (key, spec, prepared) in enumerate(prepared_entries):
+            hypothesis = prepared["hypothesis"]
+            context_ref = receipt["context_ref"]
+            # 分支隔离:带自身要求/依赖的假设派生专属子上下文
+            # (derive_child_context 拷贝父事实;分支互不污染)
+            if spec.get("requirements") or spec.get("dependencies") \
+                    or spec.get("assumptions"):
+                child_ctx = self._derive_branch_context(
+                    parent_ctx, prepared, hypothesis.hypothesis_id)
+                context_ref = persist_context(
+                    self.ledger.artifact_store, child_ctx)
+            rank = int(rank_by_key.get(key, index + 1))
+            task_key = expansion_task_key(
+                run_id=self.run_id,
+                parent_path_id=receipt["parent_path_id"],
+                context_ref=context_ref,
+                hypothesis_id=hypothesis.hypothesis_id, attempt_id="",
+                policy_version=receipt["policy_version"],
+                strict_state_digest=substrate.state_id,
+                tool_id="atom-tool-vendored",
+                prompt_version=_plan_digest(self.plan)[:16],
+                feedback_policy_version=FORMAT_FEEDBACK_POLICY_VERSION)
+            entries.append({
+                "hypothesis_key": key,
+                "hypothesis_id": hypothesis.hypothesis_id,
+                "rank": rank,
+                "payload": {
+                    "task_key": task_key,
+                    "spec": dict(spec),
+                    "hypothesis_key": key,
+                    "substrate_smiles": substrate_smiles,
+                    "substrate_state_ref": substrate.state_id,
+                    "context_ref": context_ref,
+                    "parent_path_id": receipt["parent_path_id"],
+                    "depth": expand_depth,
+                    "source_event_id": str(
+                        payload.get("source_event_id") or ""),
+                },
+            })
+        schedule = schedule_hypotheses(
+            self.ledger, run_id=self.run_id,
+            parent_path_id=receipt["parent_path_id"],
+            context_ref=receipt["context_ref"],
+            parent_search_revision=int(
+                receipt["parent_search_revision"]),
+            pool=entries, policy_version=receipt["policy_version"],
+            depth=expand_depth,
+            source_event_id=str(payload.get("source_event_id") or ""),
+            budget=self.plan.get("schedule_budget"))
+        _complete_action(
+            self.ledger, receipt["action_id"], worker_id=self.worker_id,
+            outcome={"kind": OUTCOME_KIND_SCHEDULED,
+                     "detail": {"pool_digest": schedule["pool_digest"],
+                                "selected": len(schedule["selected"]),
+                                "deferred": len(schedule["deferred"]),
+                                "blocked": len(schedule["blocked"]),
+                                "model": {
+                                    k: v for k, v in model_note.items()
+                                    if k != "rank_by_key"}}})
+
+    def _derive_branch_context(self, parent_ctx, prepared: dict,
+                               hypothesis_id: str):
+        from .path_context import build_requirement
+        step_requirements = []
+        for raw in (prepared["hypothesis"].requirements or []):
+            if isinstance(raw, dict):
+                fields = {k: v for k, v in dict(raw).items()
+                          if k in ("key", "level", "value", "basis",
+                                   "value_basis", "source_refs",
+                                   "applicable_dimensions",
+                                   "observed_source_conditions", "note")}
+                step_requirements.append(
+                    build_requirement(**fields))
+            else:
+                step_requirements.append(raw)
+        return derive_child_context(
+            parent_ctx,
+            step_requirements=None if not step_requirements
+            else step_requirements,
+            step_dependencies=prepared["dependencies"] or None,
+            step_edge_ref=hypothesis_id)
+
+    def _model_proposals(self, receipt: dict,
+                         prepared_entries: list) -> dict:
+        """scripted 模型 proposals[] 协议(网络 0;真实 provider 默认
+        off —— 登记被预算拒绝,网络发送被阻止)。"""
+        action_id = receipt["action_id"]
+        attempt_id = (receipt.get("attempt_refs") or [""])[0]
+        allowed = {prepared["hypothesis"].hypothesis_id
+                   for _, _, prepared in prepared_entries}
+        response = self._next_model_response()
+        request = self.ledger.register_model_request(
+            run_id=self.run_id, action_id=action_id,
+            attempt_id=attempt_id, provider_kind=self.model_provider_kind,
+            request_key=f"decide:{action_id}", purpose="proposals",
+            payload={"response": str(response)[:2000]})
+        if self.crash_after == "after_request_registered":
+            raise SearchCrashInjection(
+                "请求已登记、无响应(B2 注入点)",
+                {"action_id": action_id,
+                 "request_id": request["request_id"]})
+        note: dict = {"request_id": request["request_id"],
+                      "provider_kind": self.model_provider_kind,
+                      "network_request_count": request.get(
+                          "network_request_count", 0)}
+        try:
+            parsed = parse_model_proposals(
+                response, allowed_hypothesis_ids=allowed)
+        except ModelProtocolError as exc:
+            kind = exc.detail.get("kind")
+            if kind in FEEDBACK_ELIGIBLE:
+                # 有界格式反馈(默认 0 轮 ⇒ 反馈被预算拒 ⇒ 结构失败
+                # 保持;每轮持久记录输入/错误类别/反馈版本/次数)
+                try:
+                    feedback = register_format_feedback_round(
+                        self.ledger, run_id=self.run_id,
+                        action_id=action_id, attempt_id=attempt_id,
+                        error=exc, response_excerpt=str(response),
+                        provider_kind=PROVIDER_KIND_SCRIPTED)
+                except SearchBudgetExhausted as feedback_exhausted:
+                    # 反馈轮超上限(或策略 0):结构失败保持,不以反馈
+                    # 重试到通过 —— 升级为结构失败(不是真实请求预算
+                    # 的资源拒绝路径)
+                    note["format_feedback_exhausted"] = {
+                        "kind": feedback_exhausted.detail.get("kind")}
+                    raise PPV1RunError(
+                        "模型响应结构失败(格式反馈耗尽或策略 0 轮;"
+                        "结构失败保持,不以反馈改分子再试到通过)",
+                        {"kind": "model_structure_failure",
+                         "error_kind": kind,
+                         "feedback_kind":
+                             feedback_exhausted.detail.get("kind")})                         from feedback_exhausted
+                note["format_feedback_round"] = feedback["request_id"]
+                retry = self._next_model_response()
+                parsed = parse_model_proposals(
+                    retry, allowed_hypothesis_ids=allowed)
+            else:
+                # structure_unparseable:结构失败保持(不进格式反馈,
+                # 不能以反馈改分子再试到通过)
+                note["structure_failure"] = {"error_kind": kind}
+                raise PPV1RunError(
+                    "模型响应结构失败(不可解析;结构失败保持,不进格式"
+                    "反馈)", {"kind": "model_structure_failure",
+                              "error_kind": kind}) from exc
+        # rank 提供稳定次序(模型选择不升成 recorded)
+        key_by_id = {prepared["hypothesis"].hypothesis_id: key
+                     for key, _, prepared in prepared_entries}
+        rank_by_key = {}
+        for position, hyp_id in enumerate(parsed["rank_hypotheses"], 1):
+            key = key_by_id.get(hyp_id)
+            if key is not None:
+                rank_by_key[key] = position
+        note["parsed_proposals"] = len(parsed["proposals"])
+        note["abstained"] = parsed["abstained"]
+        note["rank_by_key"] = rank_by_key
+        return note
+
+    def _next_model_response(self) -> str:
+        if self.model_responses:
+            return str(self.model_responses.pop(0))
+        return "{}"
+
+    # ------------------------------------------------------------ expand
+
+    def _run_expand(self, receipt: dict) -> None:
+        payload = receipt["payload"]
+        action_id = receipt["action_id"]
+        attempt_id = (receipt.get("attempt_refs") or [""])[0]
+        substrate = _substrate_state(
+            str(payload.get("substrate_smiles") or ""),
+            f"ppv1-substrate:{self.run_id}")
+        spec = dict(payload.get("spec") or {})
+        prepared = prepare_hypothesis(
+            spec, parent_path_id=receipt["parent_path_id"],
+            substrate_state=substrate,
+            transformation_ref=str(spec.get("transformation_ref") or ""))
+        context = load_context(self.ledger.artifact_store,
+                               receipt["context_ref"])
+        limits = self.ledger._limits(self.run_id)
+        write_bundle = self.crash_after == "after_bundle_written"
+        try:
+            production = expand_hypothesis_production(
+                self.ledger, action_id, prepared=prepared, context=context,
+                worker_id=self.worker_id, attempt_id=attempt_id,
+                task_key=str(payload.get("task_key") or ""),
+                substrate_state=substrate, limits=limits,
+                depth=int(payload.get("depth") or 1),
+                parent_search_revision=int(
+                    receipt["parent_search_revision"]),
+                write_bundle=write_bundle,
+                crash_after=("bundle_written" if write_bundle else
+                             ("event_submitted"
+                              if self.crash_after == "after_event_submitted"
+                              else None)))
+        except AuditContractError as exc:
+            # **窄捕获(R8 P2 ⑧ 义务)**:契约异常中止本次
+            # admission/提交 —— 生产链在提交前上抛 ⇒ 零事件/零提交;
+            # 此处只记方法失败并调度独立备选,**不吞成成功**(无事件/
+            # 无提交/无 verified 可言;attempt 落 failed)。
+            record_expansion_outcome(
+                self.ledger, attempt_id,
+                outcome={"kind": OUTCOME_KIND_CONTRACT_ABORTED,
+                         "detail": {"error": "AuditContractError",
+                                    "reason": str(exc)[:300]}},
+                evidence_refs=[action_id], worker_id=self.worker_id)
+            return
+        status = production.get("status")
+        if status == "method_failed":
+            record_expansion_outcome(
+                self.ledger, attempt_id,
+                outcome={"kind": OUTCOME_KIND_METHOD_FAILED,
+                         "detail": {"execution": {
+                             "ok": production["execution"]["execution_ok"],
+                             "reasons": production["execution"]["result"]
+                             .get("reasons", [])[:5]}}},
+                evidence_refs=[action_id], worker_id=self.worker_id)
+            return
+        if status == "rejected":
+            record_expansion_outcome(
+                self.ledger, attempt_id,
+                outcome={"kind": OUTCOME_KIND_ADMISSION_REJECTED,
+                         "detail": {"admission":
+                                    production["admission"]}},
+                evidence_refs=[action_id], worker_id=self.worker_id)
+            return
+        if status == "deferred":
+            # R3/R4:假设保留(put_hypothesis 留痕;不提交事件;当前
+            # 场景不展开,可展示为需不同条件的备选)
+            self.store.put_hypothesis(
+                prepared["hypothesis"].to_dict())
+            record_expansion_outcome(
+                self.ledger, attempt_id,
+                outcome={"kind": OUTCOME_KIND_ADMISSION_DEFERRED,
+                         "detail": {"admission":
+                                    production["admission"]}},
+                evidence_refs=[action_id], worker_id=self.worker_id)
+            return
+        # submitted:followup 调度(场景转换优先:转换须先执行,followup
+        # 才用新上下文 —— 只改变后续阶段,不重写旧边评估)
+        self.events_submitted.append(production["event_id"])
+        parent_path_id = receipt["parent_path_id"]
+        revision = int(receipt["parent_search_revision"]) + 1
+        plan_transition = self._match_transition(payload)
+        product_refs = [s.state_id for s in prepared["product_states"]]
+        substrate_smiles_by_ref = {
+            s.state_id: s.smiles_isomeric
+            for s in prepared["product_states"]}
+        if plan_transition is not None:
+            # 转换动作携带 pending followup(转换执行后由新上下文调度)
+            transition_payload = {
+                "transition": dict(plan_transition),
+                "pending_followup": {
+                    "event_id": production["event_id"],
+                    "expansion_plan": production["expansion_plan"],
+                    "product_state_refs": product_refs,
+                    "parent_path_id": parent_path_id,
+                    "context_ref": receipt["context_ref"],
+                    "parent_search_revision": revision,
+                    "depth": int(payload.get("depth") or 1)},
+            }
+            enqueue_scenario_transition(
+                self.ledger, run_id=self.run_id,
+                parent_path_id=parent_path_id,
+                context_ref=receipt["context_ref"],
+                transition=dict(plan_transition),
+                policy_version=receipt["policy_version"],
+                parent_search_revision=revision,
+                task_key=_task_key_for(
+                    run_id=self.run_id, parent_path_id=parent_path_id,
+                    context_ref=receipt["context_ref"],
+                    subject_ref=production["event_id"],
+                    kind="scenario_transition", revision=revision,
+                    policy_version=receipt["policy_version"],
+                    extra=content_hash(plan_transition)))
+            # 把 pending_followup 并入转换 action 冻结载荷(幂等)
+            self._attach_followup_to_transition(
+                parent_path_id, receipt["context_ref"], revision,
+                content_hash(plan_transition), transition_payload)
+        else:
+            enqueue_coproduct_followups(
+                self.ledger, run_id=self.run_id,
+                parent_path_id=parent_path_id,
+                context_ref=receipt["context_ref"],
+                event_id=production["event_id"],
+                product_state_refs=product_refs,
+                expansion_plan=production["expansion_plan"],
+                policy_version=receipt["policy_version"],
+                parent_search_revision=revision,
+                substrate_smiles_by_ref=substrate_smiles_by_ref,
+                depth=int(payload.get("depth") or 1))
+
+    def _attach_followup_to_transition(self, parent_path_id: str,
+                                        context_ref: str, revision: int,
+                                        transition_digest: str,
+                                        transition_payload: dict) -> None:
+        """把 pending_followup 并入转换 action 的冻结载荷(同键同内容
+        幂等;转换执行后按新上下文调度 followup)。"""
+        action_id = stable_id(
+            "sact", self.run_id, parent_path_id, context_ref,
+            stable_id("strans", parent_path_id, context_ref,
+                      transition_digest),
+            ACTION_KIND_SCENARIO_TRANSITION, str(int(revision)),
+            search_policy_version())
+        row = self.ledger.conn.execute(
+            "SELECT payload FROM search_actions WHERE action_id = ?",
+            (action_id,)).fetchone()
+        if row is None:
+            raise PPV1RunError(
+                f"转换 action {action_id} 不存在(无法附加 pending_"
+                "followup)", {"kind": "transition_action_missing"})
+        stored = json.loads(row["payload"])
+        if "pending_followup" in stored:
+            return  # 幂等:已携带
+        stored["pending_followup"] = transition_payload["pending_followup"]
+        self.ledger._begin()
+        try:
+            self.ledger.conn.execute(
+                "UPDATE search_actions SET payload = ? WHERE action_id = ?"
+                " AND status = ?",
+                (json.dumps(stored, ensure_ascii=False, sort_keys=True), action_id, "pending"))
+            self.ledger.conn.commit()
+        except Exception:
+            self.ledger.conn.rollback()
+            raise
+
+    def _match_transition(self, payload: dict) -> dict | None:
+        key = str(payload.get("hypothesis_key") or "")
+        for entry in (self.plan.get("scenario_transitions") or []):
+            if str(entry.get("after_hypothesis_key") or "") == key:
+                return dict(entry.get("transition") or {})
+        return None
+

+
+    def _enqueue_activation(self, directive: dict) -> None:
+        payload = {
+            "task_key": _task_key_for(
+                run_id=self.run_id, parent_path_id=self.root_path_id,
+                context_ref="", subject_ref=content_hash(directive),
+                kind="activate", revision=0,
+                policy_version=search_policy_version()),
+            "deferred_type": str(directive.get("deferred_type") or ""),
+            "trigger": str(directive.get("trigger")
+                           or DEFERRED_TRIGGER_EXPLICIT),
+            "expected_parent_revision": int(
+                directive.get("expected_parent_revision") or 0),
+            "parent": str(directive.get("parent") or "root"),
+        }
+        self.ledger.enqueue_action(
+            run_id=self.run_id, parent_path_id=self.root_path_id,
+            context_ref=str(directive.get("context_ref") or ""),
+            subject_ref=stable_id("sactdef", content_hash(directive)),
+            action_kind=ACTION_KIND_ACTIVATE_DEFERRED,
+            parent_search_revision=0,
+            policy_version=search_policy_version(), payload=payload)
+
+    def _run_activate(self, receipt: dict) -> None:
+        payload = receipt["payload"]
+        deferred_type = str(payload.get("deferred_type") or "") or None
+        trigger = str(payload.get("trigger") or DEFERRED_TRIGGER_EXPLICIT)
+        expected_revision = int(payload.get("expected_parent_revision")
+                                or 0)
+        parent = str(payload.get("parent") or "root")
+        parent_path_id = (self.root_path_id if parent == "root"
+                          else parent)
+        result = activate_deferred(
+            self.ledger, run_id=self.run_id, parent_path_id=parent_path_id,
+            trigger=trigger, expected_parent_revision=expected_revision,
+            deferred_type=deferred_type)
+        _complete_action(
+            self.ledger, receipt["action_id"], worker_id=self.worker_id,
+            outcome={"kind": OUTCOME_KIND_ACTIVATED,
+                     "detail": {"activated": result["activated_ids"],
+                                "unchanged": len(result["unchanged"]),
+                                "conflicts": len(result["conflicts"]),
+                                "deferred_type": deferred_type,
+                                "trigger": trigger}})
+
+    # ------------------------------------------------------------ transition
+
+    def _run_transition(self, receipt: dict) -> None:
+        payload = receipt["payload"]
+        transition = dict(payload.get("transition") or {})
+        parent_ctx = load_context(self.ledger.artifact_store,
+                                  receipt["context_ref"])
+        child_ctx = derive_child_context(
+            parent_ctx, scenario_transition=transition,
+            step_edge_ref=receipt["action_id"])
+        context_ref = persist_context(self.ledger.artifact_store, child_ctx)
+        pending_followup = payload.get("pending_followup")
+        if isinstance(pending_followup, dict) and pending_followup:
+            # 转换已执行:followup 用新上下文调度(仅改变后续阶段)
+            followup = dict(pending_followup)
+            product_refs = list(followup.get("product_state_refs") or [])
+            smiles_by_ref = {
+                ref: self._state_smiles(ref) for ref in product_refs}
+            enqueue_coproduct_followups(
+                self.ledger, run_id=self.run_id,
+                parent_path_id=str(followup.get("parent_path_id")
+                                   or receipt["parent_path_id"]),
+                context_ref=context_ref,
+                event_id=str(followup.get("event_id") or ""),
+                product_state_refs=product_refs,
+                expansion_plan=dict(followup.get("expansion_plan") or {}),
+                policy_version=receipt["policy_version"],
+                parent_search_revision=int(
+                    followup.get("parent_search_revision")
+                    or receipt["parent_search_revision"]),
+                substrate_smiles_by_ref=smiles_by_ref,
+                depth=int(followup.get("depth") or 1))
+        _complete_action(
+            self.ledger, receipt["action_id"], worker_id=self.worker_id,
+            outcome={"kind": OUTCOME_KIND_TRANSITION_APPLIED,
+                     "detail": {"new_context_ref": context_ref,
+                                "new_phase_id": child_ctx.phase_id,
+                                "new_scenario_ref": child_ctx.scenario_ref,
+                                "source_ref": transition.get("source_ref"),
+                                "note": ("仅改变后续阶段;旧边评估冻结"
+                                         "于 phase_history,不重写")}},
+            evidence_refs=[str(transition.get("source_ref") or "")])
+
+    def _state_smiles(self, state_ref: str) -> str:
+        state = self.store.get_state(state_ref)
+        if state is None:
+            raise PPV1RunError(
+                f"产物状态 {state_ref} 不在账本(followup 调度需要产物"
+                "结构;不猜)", {"kind": "product_state_missing",
+                              "state_ref": state_ref})
+        return str(state.get("smiles_isomeric") or "")
+
+    # ------------------------------------------------------------ 停止
+
+    def _stop_report(self) -> dict:
+        no_local_candidate = (self.total_retrievals > 0
+                              and self.empty_pool_retrievals ==
+                              self.total_retrievals)
+        report = self.ledger.stop_report(
+            self.run_id, refusals=self.refusals,
+            no_local_candidate=no_local_candidate)
+        report["events_submitted_session"] = list(self.events_submitted)
+        report["runflow_version"] = RUNFLOW_VERSION
+        if self.run_error is not None:
+            report["run_error"] = dict(self.run_error)
+        return report

+    if model is not None:

+    from ..domain.policy import PREDICTION_POLICY_VERSION

+            PREDICTION_POLICY_VERSION, "policy": {}}

+def start_ppv1_run(
+    *,
+    config: V43Config,
+    runs_root: str | Path,
+    run_id: str,
+    smiles: str,
+    plan: dict | None = None,
+    model=None,
+    profile_policy: dict | None = None,
+    scripted: bool = False,
+    crash_after: str | None = None,
+) -> dict:
+    """PPV1 独立 start(分流目标;不落 legacy 图/存储)。
+
+    - profile 策略装配(build_profile_identity_block:十上限 +
+      format_feedback_policy;未知取值 fail-closed);
+    - run identity(代码/资产/策略/模型/MoleCode;PPV1 哈希全树)落盘
+      后进驱动;重复 run 拒绝;
+    - 严格 root state(工厂守卫)+ 根 context(artifact 持久化)+
+      搜索 run 注册(policy block = profile 块;恢复按同一身份);
+    - 根 retrieve 入队(计划池;空池 ⇒ no_local_candidate 诚实停止);
+    - 驱动循环 + stop_report(停止不是化学终点)。
+    """
+    runs_root = Path(runs_root)
+    run_dir = runs_root / run_id
+    if (run_dir / "run_identity.json").is_file():
+        from ..workflow.runner import RunAlreadyExists
+        raise RunAlreadyExists(run_id, run_dir)
+    # 原子独占目录创建:并发 start 同一 run 只有胜者(败者
+    # FileExistsError → RunAlreadyExists;不进 store/identity 写入)
+    runs_root.mkdir(parents=True, exist_ok=True)
+    try:
+        run_dir.mkdir()
+    except FileExistsError:
+        from ..workflow.runner import RunAlreadyExists
+        raise RunAlreadyExists(run_id, run_dir) from None
+    policy_block = build_profile_identity_block(**(profile_policy or {}))
+    limits = policy_block["search_limits"]
+    asset_identity = (scripted_asset_identity() if scripted
+                      else retrieval_asset_identity(config))
+    identity = build_run_identity(
+        run_id=run_id, root_smiles=smiles, root_cn=_cn_of(smiles),
+        strategy=_strategy_block(limits), asset_identity=asset_identity,
+        v3_code_identity=v3_code_identity(config.v3_code_root),
+        prediction=_prediction_block(model),
+        profile=PROFILE_POLLUTANT_PATHWAY_V1,
+        profile_policy=profile_policy or None)
+    write_run_identity(runs_root, identity)
+    store = PollutantPathwayStore(run_dir / "pollutant.sqlite")
+    artifacts = ArtifactStore(runs_root, run_id)
+    ledger = SearchLedger(store, artifacts, run_id)
+    ledger.register_search_run(run_id, policy_block)
+    root_state = _substrate_state(smiles, "ppv1-root-input")
+    store.put_state(root_state.to_dict())
+    root_ctx = build_path_context(
+        condition_facts=_fact_list((plan or {}).get("condition_facts")),
+        scenario_ref=str((plan or {}).get("scenario_ref") or ""),
+        dependencies=[
+            build_cyclic_dependency(dict(d))
